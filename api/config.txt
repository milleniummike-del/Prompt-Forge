<?php
// RENAME THIS FILE TO: config.php

// ============================================
// DATABASE CONFIGURATION
// ============================================

// On most shared hosts, you must create the database manually via your control panel (cPanel, etc.)
// Then fill in the details below.

$host = getenv('DB_HOST') ?: 'localhost';
$user = getenv('DB_USER') ?: 'YOUR_DB_USERNAME';
$pass = getenv('DB_PASSWORD') ?: 'YOUR_DB_PASSWORD';
$dbname = getenv('DB_NAME') ?: 'YOUR_DB_NAME';
$charset = 'utf8mb4';

$options = [
    PDO::ATTR_ERRMODE            => PDO::ERRMODE_EXCEPTION,
    PDO::ATTR_DEFAULT_FETCH_MODE => PDO::FETCH_ASSOC,
    PDO::ATTR_EMULATE_PREPARES   => false,
];

try {
    $dsn = "mysql:host=$host;dbname=$dbname;charset=$charset";
    $pdo = new PDO($dsn, $user, $pass, $options);
} catch (\PDOException $e) {
    // Return a JSON error so the frontend handles it gracefully
    header('HTTP/1.1 500 Internal Server Error');
    header("Content-Type: application/json");
    echo json_encode([
        'error' => 'Database connection failed', 
        'details' => $e->getMessage()
    ]);
    exit;
}

return $pdo;
?>
